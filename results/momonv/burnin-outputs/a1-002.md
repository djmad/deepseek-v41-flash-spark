# task 5: Write a Python class for a thread-safe bounded queue using threading.Condition.

```python
import threading
from collections import deque


class BoundedQueue:
    def __init__(self, maxsize):
        if maxsize <= 0:
            raise ValueError("maxsize must be positive")
        self.maxsize = maxsize
        self._queue = deque()
        self._cond = threading.Condition()

    def put(self, item, timeout=None):
        with self._cond:
            if not self._cond.wait_for(lambda: len(self._queue) < self.maxsize, timeout):
                raise TimeoutError("put timed out")
            self._queue.append(item)
            self._cond.notify_all()

    def get(self, timeout=None):
        with self._cond:
            if not self._cond.wait_for(lambda: len(self._queue) > 0, timeout):
                raise TimeoutError("get timed out")
            item = self._queue.popleft()
            self._cond.notify_all()
            return item

    def qsize(self):
        with self._cond:
            return len(self._queue)

    def empty(self):
        with self._cond:
            return not self._queue

    def full(self):
        with self._cond:
            return len(self._queue) >= self.maxsize
```

The `threading.Condition` guards all state access, and `wait_for` atomically blocks until the queue has space (for `put`) or an item (for `get`), with `notify_all` waking both producers and consumers.
