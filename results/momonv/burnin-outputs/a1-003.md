# task 6: Write a Python function that topologically sorts a dependency dict and raises on cycles.

```python
from collections import deque

def topological_sort(graph: dict) -> list:
    """Return a topological ordering of graph (dict: node -> iterable of deps).

    Raises ValueError on cycles. Nodes with no deps may appear as keys or only
    as values; all are included in the output.
    """
    indegree = {}
    for node, deps in graph.items():
        indegree.setdefault(node, 0)
        for dep in deps:
            indegree.setdefault(dep, 0)
            indegree[node] += 1

    queue = deque(n for n, d in indegree.items() if d == 0)
    order = []
    while queue:
        node = queue.popleft()
        order.append(node)
        for dependent, deps in graph.items():
            if node in deps:
                indegree[dependent] -= 1
                if indegree[dependent] == 0:
                    queue.append(dependent)

    if len(order) != len(indegree):
        raise ValueError("Cycle detected in dependency graph")
    return order
```

It builds an in-degree map over all nodes (including those appearing only as dependencies), peels off zero-in-degree nodes with BFS, and raises `ValueError` if the processed count falls short of the total node count.
